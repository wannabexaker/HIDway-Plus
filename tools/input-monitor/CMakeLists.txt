add_executable(hidway-input-monitor WIN32 main.c model.c device.c observer.c monitor.rc)
add_executable(hidway-monitor-tests ../../tests/test_monitor_win.c model.c device.c)
foreach(t hidway-input-monitor hidway-monitor-tests)
    target_include_directories(${t} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_definitions(${t} PRIVATE UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS WINVER=0x0A00 _WIN32_WINNT=0x0A00)
    target_link_libraries(${t} PRIVATE user32 gdi32 comctl32 setupapi cfgmgr32 uxtheme)
    if(MSVC)
        target_compile_options(${t} PRIVATE /W4 /WX /utf-8)
        set_property(TARGET ${t} PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
    endif()
endforeach()
target_link_libraries(hidway-input-monitor PRIVATE dwmapi)
if(MSVC)
    target_link_options(hidway-input-monitor PRIVATE /MANIFEST:NO)
endif()
add_test(NAME hidway_monitor COMMAND hidway-monitor-tests)
